<script>
function calculateSalary() {
    // 1. Fetch values from input fields
    const basicPayInput = document.getElementById('basicPay').value;
    
    // Validation check
    if (basicPayInput === '' || basicPayInput < 0) {
        alert('Please enter a valid Basic Pay.');
        return;
    }
    
    const basicPay = parseFloat(basicPayInput);

    // 2. Define salary calculation percentages
    const daRate = 0.40;   // 40% of Basic Pay
    const hraRate = 0.20;  // 20% of Basic Pay
    const pfRate = 0.12;   // 12% of Basic Pay
    const taxRate = 0.05;  // 5% of Basic Pay

    // 3. Perform calculations
    const da = basicPay * daRate;
    const hra = basicPay * hraRate;
    const grossPay = basicPay + da + hra;
    
    const pf = basicPay * pfRate;
    const tax = basicPay * taxRate;
    const deduction = pf + tax;
    
    const netPay = grossPay - deduction;

    // 4. Update the read-only form fields with formatted numbers (2 decimal places)
    document.getElementById('da').value = da.toFixed(2);
    document.getElementById('hra').value = hra.toFixed(2);
    document.getElementById('grossPay').value = grossPay.toFixed(2);
    document.getElementById('pf').value = pf.toFixed(2);
    document.getElementById('tax').value = tax.toFixed(2);
    document.getElementById('deduction').value = deduction.toFixed(2);
    document.getElementById('netPay').value = netPay.toFixed(2);
}
</script>
